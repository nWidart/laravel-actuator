import { Card, CardGrid, LinkCard } from '@astrojs/starlight/components';

```bash
composer require nwidart/laravel-actuator
curl http://localhost/actuator/health
# {"status":"UP","groups":["liveness","readiness"]}
```

<CardGrid stagger>
	<Card title="Health that orchestrators understand" icon="approve-check">
		Status aggregation, HTTP 503 for `DOWN` and `OUT_OF_SERVICE`, and ready-made `liveness` and
		`readiness` groups for Kubernetes probes.
	</Card>
	<Card title="Built-in indicators" icon="puzzle">
		Database, cache, Redis, disk space and maintenance mode. Write your own in one class.
	</Card>
	<Card title="Spring Boot Actuator format" icon="random">
		`health`, `info`, `env` and `scheduledtasks` use the Spring Boot JSON, so tools such as Spring
		Boot Admin can read them.
	</Card>
	<Card title="Metrics for Prometheus" icon="rocket">
		Requests, queued jobs, database queries, exceptions and scheduled tasks, ready to scrape. Add
		your own counters and timers.
	</Card>
	<Card title="Safe by default" icon="warning">
		Only `health` is exposed, and it hides its details. Add middleware and a gate before you open
		anything else.
	</Card>
</CardGrid>
